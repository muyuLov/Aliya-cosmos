**Step 3: 最小实现**

```ts
// emotion-map.ts
export const EMOTION_DEFAULT = "none" as const;

const MAP: Record<string, string> = {
  joy: "squint",
  happy: "squint",
  excited: "squint",
  love: "blush",
  like: "blush",
  shy: "shy",
  embarrassed: "shy",
  surprise: "shocked",
  amazed: "shocked",
  sad: "dark",
  sorrow: "dark",
  lonely: "dark",
  angry: "dark",
  fear: "shocked",
  daze: "daze",
  tired: "daze",
  // 其余标签回退默认
};

export function emotionToExpression(dominant: string): string {
  return MAP[dominant] ?? EMOTION_DEFAULT;
}
```

> "none" 表示不切换表情（保持默认）。

**Step 5: 桌宠 main.ts 接入**

在 `manager.onLoad` 回调后订阅情绪事件：

```ts
// main.ts（追加）
let lastExpression = "";
const emotionOff = window.live2dAction?.onPlayAction?.((target) => {
  void manager.playAction(target);
}) ?? (() => {});
// 订阅情绪 → 表情（经主进程 state:changed 或 agui EMOTION_CHANGED）
const applyEmotion = (dominant: string): void => {
  const expr = emotionToExpression(dominant);
  if (expr === "none") {
    if (lastExpression) { void manager.playAction({ kind: "expression", name: lastExpression }); } // 复原由 expression-reset 处理
    return;
  }
  lastExpression = expr;
  void manager.playAction({ kind: "expression", name: expr });
};
```

> 事件来源：P2 主进程已广播 `state:changed`（含 emotion）。桌宠通过 preload 订阅 `window.runtimeState.onChanged`（P3 若已提供）。若尚未提供，在 preload 补 `runtimeState` 桥（`state:changed` 转发）。

**Step 1: preload 补 audio 桥**

```ts
// preload/index.ts（追加）
contextBridge.exposeInMainWorld("audioBridge", {
  onChunk: (cb: (buf: ArrayBuffer) => void) => {
    const l = (_e: unknown, buf: ArrayBuffer) => cb(buf);
    ipcRenderer.on("audio:chunk", l);
    return () => ipcRenderer.off("audio:chunk", l);
  },
});
```

**Step 2: 桌宠嘴型驱动**

在桌宠 `main.ts` 中：

```ts
// main.ts（追加）
let audioCtx: AudioContext | null = null;
let mouthTimer: ReturnType<typeof setTimeout> | null = null;

window.audioBridge?.onChunk((buf) => {
  audioCtx ??= new AudioContext();
  void audioCtx.decodeAudioData(buf).then((audioBuf) => {
    const src = audioCtx!.createBufferSource();
    src.buffer = audioBuf;
    src.connect(audioCtx!.destination);
    const durationMs = audioBuf.duration * 1000;
    // 嘴型：duration 期间开启，播完关闭
    if (mouthTimer) clearTimeout(mouthTimer);
    mouthSync?.start(durationMs);
    speakingMotion?.start();
    src.onended = () => {
      mouthSync?.stop();
      speakingMotion?.stop();
    };
    src.start();
  }).catch(() => { /* 解码失败静默 */ });
});
```

> `mouthSync` / `speakingMotion` 变量已在桌宠 `main.ts` 中定义（P1 拷贝）。后端推的音频是 PCM/MP3 帧，若 `decodeAudioData` 失败（MP3 需要 codec），改用 `AudioContext.decodeAudioData` 兼容或播放原始 PCM（`BufferSource` + PCM 参数配置）。实测为准。
